cmake_minimum_required(VERSION 3.24)
project(Practice7 LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_executable(Practice7 main.cpp)
target_compile_features(Practice7 PRIVATE cxx_std_17)
target_link_libraries(Practice7 PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(Practice7 PRIVATE /utf-8)
endif()
foreach(number RANGE 1 3)
  configure_file(traffic_light${number}.json traffic_light${number}.json COPYONLY)
endforeach()
set_property(TARGET Practice7 PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
